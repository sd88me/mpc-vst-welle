cmake_minimum_required(VERSION 3.15)
project(wave_mpc VERSION 0.1.0 LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
# Everything hidden, no STB_GNU_UNIQUE symbols: the .so must be unloadable and export nothing but VSTPluginMain.
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fvisibility=hidden -fvisibility-inlines-hidden -fno-gnu-unique")
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -fvisibility=hidden")
if(NOT MPC_VST)
  set(MPC_VST $ENV{HOME}/mpc-vst-plugins)
endif()

set(TP ${CMAKE_CURRENT_SOURCE_DIR}/third_party)
set(W ${CMAKE_CURRENT_SOURCE_DIR}/src/wave)

# --- Musashi (MC68000), adapted the way Wave's build adapts it: thread-local execution state ---
include(cmake/PrepareMusashi.cmake)
set(MUSASHI_DIR ${CMAKE_BINARY_DIR}/musashi)
wave_prepare_musashi(${TP}/Musashi ${MUSASHI_DIR})
# m68kops.c/.h are generated by Musashi's m68kmake, built for the build host (build.sh: build/musashi-gen)
set(MUSASHI_GEN ${CMAKE_CURRENT_SOURCE_DIR}/build/musashi-gen)
if(NOT EXISTS ${MUSASHI_GEN}/m68kops.c)
  message(FATAL_ERROR "run ./build.sh musashi first (generates build/musashi-gen/m68kops.c)")
endif()
add_library(WaveM68k STATIC ${MUSASHI_DIR}/m68kcpu.c ${MUSASHI_DIR}/softfloat/softfloat.c ${MUSASHI_GEN}/m68kops.c)
target_include_directories(WaveM68k PUBLIC ${MUSASHI_DIR} ${MUSASHI_GEN})
target_compile_definitions(WaveM68k PRIVATE WAVE_M68K_THREAD_LOCAL=_Thread_local
  M68K_EMULATE_010=0 M68K_EMULATE_EC020=0 M68K_EMULATE_020=0 M68K_EMULATE_030=0 M68K_EMULATE_040=0
  M68K_EMULATE_ADDRESS_ERROR=1 M68K_INSTRUCTION_HOOK=1)

# --- JUCE, headless: core/events/data_structures/audio_basics/audio_formats/processors_headless/dsp/cryptography ---
set(JUCE_DEFS JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_STANDALONE_APPLICATION=0 JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0
  JUCE_MODULE_AVAILABLE_juce_core=1 JUCE_MODULE_AVAILABLE_juce_events=1 JUCE_MODULE_AVAILABLE_juce_data_structures=1
  JUCE_MODULE_AVAILABLE_juce_audio_basics=1 JUCE_MODULE_AVAILABLE_juce_audio_formats=1
  JUCE_MODULE_AVAILABLE_juce_audio_processors_headless=1 JUCE_MODULE_AVAILABLE_juce_audio_processors=1
  JUCE_MODULE_AVAILABLE_juce_dsp=1 JUCE_MODULE_AVAILABLE_juce_cryptography=1
  JUCE_USE_FLAC=0 JUCE_USE_OGGVORBIS=0 JUCE_USE_MP3AUDIOFORMAT=0 JUCE_USE_LAME_AUDIO_FORMAT=0 JUCE_USE_WINDOWS_MEDIA_FORMAT=0
  JUCE_PLUGINHOST_VST3=0 JUCE_PLUGINHOST_LV2=0 JUCE_PLUGINHOST_LADSPA=0 JUCE_PLUGINHOST_ARA=0 JUCE_DSP_USE_STATIC_FFTW=0
  JUCE_CHECK_MEMORY_LEAKS=0 JUCE_DISPLAY_SPLASH_SCREEN=0 JUCE_LOG_ASSERTIONS=0 NDEBUG=1
  JucePlugin_Name="Wave" JucePlugin_IsSynth=1 JucePlugin_WantsMidiInput=1 JucePlugin_ProducesMidiOutput=0)
file(GLOB JM src/gen/jm_*.cpp)
add_library(wave_juce STATIC ${JM} src/shim/juce_audio_processors/apvts.cpp
  src/shim/juce_audio_utils/juce_MidiMessageCollector.cpp)
target_include_directories(wave_juce PUBLIC src/shim ${TP}/JUCE/modules)
target_compile_definitions(wave_juce PUBLIC ${JUCE_DEFS})

# --- Wave Emulation (vendored in src/wave, see src/VENDORED.md) ---
set_source_files_properties(cmake/assets.S PROPERTIES COMPILE_DEFINITIONS
  "WAVE_BLANK_IMG=\"${CMAKE_CURRENT_SOURCE_DIR}/src/wave-data/BlankWave.img\";WAVE_PPG_BIN=\"${CMAKE_CURRENT_SOURCE_DIR}/src/wave-data/ppg-v6-wavetables.bin\"")
file(GLOB WAVE_SRC ${W}/Dsp/*.cpp ${W}/Firmware/*.cpp ${W}/Presets/*.cpp ${W}/UI/*.cpp ${W}/*.cpp)
add_library(WaveCore STATIC ${WAVE_SRC} cmake/assets.S)
# WAVE_FAST_MATH: the voice DSP with -ffast-math (not the firmware or JUCE)
if(WAVE_FAST_MATH)
  file(GLOB WAVE_DSP_SRC ${W}/Dsp/*.cpp)
  set_source_files_properties(${WAVE_DSP_SRC} PROPERTIES COMPILE_OPTIONS "-ffast-math;-fno-finite-math-only")
endif()
target_include_directories(WaveCore PUBLIC ${W} src/gen)
target_compile_definitions(WaveCore PUBLIC WAVE_HAS_PRIVATE_UPPER_TABLES=0 WAVE_HAS_PRIVATE_ROMS=0 WAVE_HAS_FACTORY_SET=0)
target_link_libraries(WaveCore PUBLIC wave_juce WaveM68k)

add_executable(wave_smoke test/smoke.cpp)
target_link_libraries(wave_smoke PRIVATE WaveCore pthread dl)
add_executable(wave_dump_params test/dump_params.cpp)
target_link_libraries(wave_dump_params PRIVATE WaveCore pthread dl)

# --- the MPC engine (src/dsp), linked with the wrapper's vst2_wrap.c into wavicle.so by build.sh ---
add_library(wavicle_dsp STATIC src/dsp/wavicle_engine.cpp)
target_include_directories(wavicle_dsp PUBLIC src/dsp ${MPC_VST}/wrapper)
target_link_libraries(wavicle_dsp PUBLIC WaveCore)

add_executable(wavicle_engine_test test/engine_test.cpp)
target_link_libraries(wavicle_engine_test PRIVATE wavicle_dsp pthread dl)
add_executable(wave_lcd_dump test/lcd_dump.cpp)
target_link_libraries(wave_lcd_dump PRIVATE WaveCore pthread dl)
add_executable(wavicle_hold_bench test/hold_bench.cpp)
target_link_libraries(wavicle_hold_bench PRIVATE wavicle_dsp pthread dl)
add_executable(wave_prof test/prof.cpp)
target_link_libraries(wave_prof PRIVATE WaveCore pthread dl)
add_executable(wave_layers test/layers.cpp)
target_link_libraries(wave_layers PRIVATE WaveCore pthread dl)
